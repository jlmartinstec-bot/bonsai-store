import sqlite3
import os

DB_NAME = "bonsai.db"

def get_connection():
    conn = sqlite3.connect(DB_NAME)
    return conn

def init_db():
    if not os.path.exists(DB_NAME):
        conn = get_connection()
        cursor = conn.cursor()

        # criando tabela manualmente, sem carregar o arquivo .sql
        cursor.execute("""
            CREATE TABLE products (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                nome TEXT,
                descricao TEXT,
                preco REAL
            )
        """)

        conn.commit()
        conn.close()
        print("Banco criado.")
