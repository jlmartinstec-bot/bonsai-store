import sqlite3
import os

DB_NAME = "bonsai.db"

# Função que cria a conexão com o banco
def get_connection():
    conn = sqlite3.connect(DB_NAME)
    return conn

# Função para inicializar o banco usando o schema.sql
def inicializar_banco():
    conn = get_connection()
    cursor = conn.cursor()

    # Lê o arquivo schema.sql
    with open("../Database/schema.sql", "r") as f:
        sql_script = f.read()

    # Executa todo o script SQL
    cursor.executescript(sql_script)

    conn.commit()
    conn.close()
